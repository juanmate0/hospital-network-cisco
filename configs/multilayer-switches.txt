! ===== Multilayer switches: base config + SSH =====
enable
configure terminal
hostname DD-MultilayerSW2
enable password cisco
banner motd #DD-MultilayerSWITCH2 CONSOLE#
no ip domain-lookup
line console 0
 password cisco
 login
end
configure terminal
service password-encryption

ip domain-name cisco.net
username admin password cisco
crypto key generate rsa
1024
line vty 0 15
 login local
 transport input ssh
end
write memory

! ===== VLANs on L3 switches (trunks toward access switches) =====
vlan 10
vlan 20
vlan 30
vlan 40
vlan 50
vlan 60

 switchport mode trunk
exit

vlan 70
vlan 80
vlan 90
vlan 100

vlan 120
